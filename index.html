function serializeGameState() { const p = gameState.player; return { version: SAVE_VERSION, savedAt: new Date().toISOString(), voiceKey: gameState.voiceKey, operaIdx: gameState.currentOperaIdx, stageIdx: gameState.currentStageIdx, health: gameState.health, score: gameState.score, time: gameState.time, levelSystem: { level: levelSystem.level, xp: levelSystem.xp, points: levelSystem.points, stats: { ...levelSystem.stats } }, player: { x: p.x, y: p.y, vx: p.vx, vy: p.vy, facing: p.facing, onGround: p.onGround }, camera: { x: gameState.camera.x, y: gameState.camera.y }, // 👇 ДОБАВЬТЕ ЭТО platforms: gameState.platforms.map(pl => ({ ...pl })), enemiesKilled: gameState.enemiesKilled, enemiesTotal: gameState.enemiesTotal, notesCollected: gameState.notesCollected, notesNeeded: gameState.notesNeeded, ariaCooldowns: { ...gameState.ariaCooldowns }, enemies: gameState.enemies .filter(e => e.alive) .map(e => ({ x: e.x, y: e.y, vx: e.vx, w: e.w, h: e.h, hp: e.hp, maxHp: e.maxHp, type: e.type, patrolMinX: e.patrolMinX, patrolMaxX: e.patrolMaxX })), boss: gameState.boss ? { x: gameState.boss.x, y: gameState.boss.y, hp: gameState.boss.hp, maxHp: gameState.boss.maxHp } : null, collectibleNotes: gameState.collectibleNotes .filter(n => !n.collected) .map(n => ({ x: n.x, y: n.y, type: n.type })), effects: { ...gameState.effects }, invulnTimer: gameState.invulnTimer }; }
